OAuth 2.0 Token ExchangeNode.js · Step 4 of auth flow
// whoop-auth.js — server-side only, never expose client_secret to the browser
async function exchangeCode(code) {
  const res = await fetch('https://api.prod.whoop.com/oauth/oauth2/token', {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: new URLSearchParams({
      grant_type: 'authorization_code', code,
      client_id:     process.env.WHOOP_CLIENT_ID,
      client_secret: process.env.WHOOP_CLIENT_SECRET,
      redirect_uri:  process.env.WHOOP_REDIRECT_URI,
    })
  });
  if (!res.ok) throw new Error(`Token exchange failed: ${res.status}`);
  return res.json(); // { access_token, refresh_token, expires_in: 3600, token_type: "Bearer" }
}

async function refreshToken(refresh) {
  const res = await fetch('https://api.prod.whoop.com/oauth/oauth2/token', {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: new URLSearchParams({
      grant_type:    'refresh_token',
      refresh_token: refresh,
      client_id:     process.env.WHOOP_CLIENT_ID,
      client_secret: process.env.WHOOP_CLIENT_SECRET,
    })
  });
  return res.json();
}
Fetching Health DataRecovery · Sleep · Workout endpoints
const BASE = 'https://api.prod.whoop.com/developer/v1';
const headers = (token) => ({ Authorization: `Bearer ${token}` });

// Recovery: HRV, RHR, SpO2, skin temp, recovery score
async function getRecovery(token, days = 7) {
  const end   = new Date().toISOString();
  const start = new Date(Date.now() - days * 86400000).toISOString();
  const res = await fetch(
    `${BASE}/recovery?limit=${days}&start=${start}&end=${end}`,
    { headers: headers(token) }
  );
  const data = await res.json();
  return data.records; // sorted desc; handle next_token for pagination
}

// Daily cycle: strain, calories, avg/max HR
async function getCycle(token) {
  const res = await fetch(`${BASE}/cycle?limit=1`, { headers: headers(token) });
  return (await res.json()).records[0];
}

// Latest sleep session with stage breakdown
async function getSleep(token) {
  const res = await fetch(`${BASE}/activity/sleep?limit=1`, { headers: headers(token) });
  return (await res.json()).records[0];
}

// Recent workouts with HR zone breakdown
async function getWorkouts(token, limit = 5) {
  const res = await fetch(`${BASE}/activity/workout?limit=${limit}`, { headers: headers(token) });
  return (await res.json()).records;
}

// Retry with exponential backoff on 429 rate-limit errors
async function whoopFetch(url, token, retries = 3) {
  for (let i = 0; i < retries; i++) {
    const res = await fetch(url, { headers: { Authorization: `Bearer ${token}` } });
    if (res.status === 401) { token = await refreshToken(savedRefreshToken); continue; }
    if (res.status === 429) { await sleep(2 ** i * 1000); continue; }
    if (!res.ok) throw new Error(`Whoop API ${res.status}`);
    return res.json();
  }
}

// Paginate through all recovery records
async function getAllRecovery(token) {
  const all = []; let nextToken = null;
  do {
    const url  = `${BASE}/recovery?limit=25` + (nextToken ? `&nextToken=${nextToken}` : '');
    const data = await whoopFetch(url, token);
    all.push(...data.records);
    nextToken = data.next_token ?? null;
  } while (nextToken);
  return all;
}

📖 README — Whoop Sample Integration App

This app demonstrates how to connect to the Whoop Developer API and visualise health metrics in a web dashboard. The dashboard runs on mock data by default; follow the steps below to connect your real Whoop account.

⚠️ No sandbox: Whoop has no test environment. You need a real Whoop device and an active subscription to obtain live data. Use the mock data mode for development and CI pipelines.

Prerequisites

1
Active Whoop membership
A Whoop 4.0 device with an active subscription ($30/month or annual). API access requires the device to sync data to Whoop's cloud.
2
Node.js 18+ installed
The backend auth proxy is a small Express server. Run node --version to check. Download from nodejs.org if needed.
3
Whoop Developer account
Register a free developer account at developer.whoop.com. Create a new application — you'll receive a client_id and client_secret.

Installation

git clone https://github.com/your-org/whoop-dashboard-spike
cd whoop-dashboard-spike
npm install

Environment Variables

Create a .env file in the project root. Never commit this file.

WHOOP_CLIENT_ID=your_client_id_here
WHOOP_CLIENT_SECRET=your_client_secret_here
WHOOP_REDIRECT_URI=http://localhost:3000/auth/callback
SESSION_SECRET=any_random_string_32_chars_or_more
PORT=3000

Running Locally

# Start the Express backend (handles OAuth + proxies API calls)
npm run dev

# Open in browser
open http://localhost:3000
1
Click "Connect Whoop Account"
You'll be redirected to api.prod.whoop.com/oauth/oauth2/auth where you log in with your Whoop credentials and grant the app permission to read your data.
2
Whoop redirects back
After consent, Whoop redirects to /auth/callback?code=.... The server exchanges the code for tokens and stores them in the session.
3
Dashboard loads real data
The frontend fetches /recovery, /api/cycle, and /api/sleep — your backend proxies these to the Whoop API using the stored token, refreshing automatically when expired.

OAuth Scopes Requested

The app requests the following scopes during the authorization step:

  • read:profile — name, email, unit preferences
  • read:recovery — HRV, RHR, SpO₂, skin temp, recovery score
  • read:cycles — daily strain and calorie data
  • read:sleep — sleep sessions and stage breakdown
  • read:workout — workout sessions and HR zone data

Mock Data Mode

By default (no token present), the app runs in mock mode — all metrics are randomly generated on each page load. To force mock mode even when a token exists:

WHOOP_MOCK_DATA=true npm run dev

Data Freshness

ℹ️ Whoop data is not real-time. Recovery, sleep, and HRV data only become available after your device syncs post-sleep (typically within 5–10 minutes of waking). The dashboard caches API responses for 15 minutes to avoid rate-limit issues.

Rate Limits

The Whoop API enforces approximately 100 requests per minute per user token. The backend implements exponential backoff on 429 responses. For production, cache responses in Redis or a database — don't fetch from Whoop on every frontend request.

Going to Production

  • Store tokens encrypted in your database, not in memory/session
  • Run token refresh on a scheduled job, not on user request
  • Never expose client_secret or user tokens to the browser
  • Implement GDPR consent before storing any biometric data
  • Add HTTPS — Whoop requires TLS for production redirect URIs
